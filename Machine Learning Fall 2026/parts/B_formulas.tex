\section{Mathematics (Chapter 2)}

\rowcolors{2}{white}{lightbg}
\begin{longtable}{@{}L{4.0cm}L{11.0cm}@{}}
\toprule
\rowcolor{primary!15}
\textbf{Quantity} & \textbf{Formula} \\
\midrule
\endhead
Dot product & $\mathbf{w}\cdot\mathbf{x} = \sum_j w_j x_j$ \\
Euclidean norm, distance & $\|\mathbf{x}\| = \sqrt{\sum_j x_j^2}$, \quad $d(\mathbf{x}, \mathbf{z}) = \|\mathbf{x} - \mathbf{z}\|$ \\
Cosine similarity & $\cos\theta = \mathbf{x}\cdot\mathbf{z} / (\|\mathbf{x}\|\,\|\mathbf{z}\|)$ \\
Derivative rules & $(w^k)' = k w^{k-1}$; $\;(f + g)' = f' + g'$; $\;(f(g(w)))' = f'(g(w))\,g'(w)$ \\
Gradient descent & $\mathbf{w} \leftarrow \mathbf{w} - \alpha\,\nabla L(\mathbf{w})$ \\
Bayes' theorem & $P(A \mid B) = P(B \mid A)P(A) / P(B)$ \\
Total probability & $P(B) = P(B \mid A)P(A) + P(B \mid \neg A)P(\neg A)$ \\
Mean, variance & $\bar{x} = \tfrac{1}{n}\sum x_i$, \quad $\operatorname{Var} = \mathbb{E}[(X - \mu)^2]$ \\
Gaussian density & $p(x) = \frac{1}{\sqrt{2\pi\sigma^2}}\exp\!\big(-\frac{(x-\mu)^2}{2\sigma^2}\big)$ \\
Entropy & $H = -\sum_i p_i\log_2 p_i$ \\
\bottomrule
\end{longtable}

\section{Workflow and Linear Models (Chapters 3, 5, 6)}

\rowcolors{2}{white}{lightbg}
\begin{longtable}{@{}L{4.0cm}L{11.0cm}@{}}
\toprule
\rowcolor{primary!15}
\textbf{Quantity} & \textbf{Formula} \\
\midrule
\endhead
Standardisation & $z = (x - \mu_{\text{train}})/\sigma_{\text{train}}$ \\
Min--max scaling & $x' = (x - \min)/(\max - \min)$ \\
Linear model & $\hat{y} = \mathbf{w}\cdot\mathbf{x} + b$ \\
Mean squared error & $\text{MSE} = \tfrac{1}{n}\sum (y_i - \hat{y}_i)^2$ \\
Least squares, one feature & $w = \dfrac{\sum (x_i - \bar{x})(y_i - \bar{y})}{\sum (x_i - \bar{x})^2}$, \quad $b = \bar{y} - w\bar{x}$ \\
Normal equation & $\boldsymbol{\theta} = (X^{\top}X)^{-1}X^{\top}\mathbf{y}$ \\
MSE gradient & $\partial\,\text{MSE}/\partial w_j = \tfrac{2}{n}\sum(\hat{y}_i - y_i)x_{ij}$ \\
MAE, RMSE, $R^2$ & $\tfrac{1}{n}\sum|y_i - \hat{y}_i|$; \quad $\sqrt{\text{MSE}}$; \quad $1 - \sum(y_i - \hat{y}_i)^2 / \sum(y_i - \bar{y})^2$ \\
Sigmoid & $\sigma(z) = 1/(1 + e^{-z})$, \quad $\sigma' = \sigma(1 - \sigma)$ \\
Log-odds, odds ratio & $z = \ln\frac{p}{1-p}$; \quad odds multiplied by $e^{w_j}$ per unit of $x_j$ \\
Cross-entropy & $L = -\tfrac{1}{n}\sum\big[y_i\ln p_i + (1 - y_i)\ln(1 - p_i)\big]$ \\
Cross-entropy gradient & $\partial L/\partial w_j = \tfrac{1}{n}\sum(p_i - y_i)x_{ij}$ \\
Softmax & $P(y = k \mid \mathbf{x}) = e^{z_k}/\sum_j e^{z_j}$ \\
\bottomrule
\end{longtable}

\begin{longtable}{@{}L{4.0cm}L{11.0cm}@{}}
\toprule
\rowcolor{primary!15}
\textbf{Quantity} & \textbf{Formula} \\
\midrule
\endhead
Bias--variance & $\mathbb{E}[(y - \hat{f})^2] = \text{bias}^2 + \text{variance} + \sigma^2_{\varepsilon}$ \\
Ridge, lasso & $\text{MSE} + \lambda\sum w_j^2$; \quad $\text{MSE} + \lambda\sum|w_j|$ \\
Ridge, one dimension & $w = \sum x_i y_i / (\sum x_i^2 + \lambda)$ \quad (centred data) \\
Accuracy & $(\text{TP} + \text{TN})/\text{all}$ \\
Precision, recall & $\text{TP}/(\text{TP} + \text{FP})$; \quad $\text{TP}/(\text{TP} + \text{FN})$ \\
Specificity, FPR & $\text{TN}/(\text{TN} + \text{FP})$; \quad $\text{FPR} = 1 - \text{specificity}$ \\
$F_1$ & $2\,\text{prec}\cdot\text{rec}/(\text{prec} + \text{rec})$ \\
95\% interval for error & $e \pm 1.96\sqrt{e(1 - e)/n}$ \\
McNemar's test & $\chi^2 = (|b - c| - 1)^2/(b + c)$; significant at 5\% if $> 3.84$ \\
\bottomrule
\end{longtable}

\section{Supervised Algorithms (Chapters 9--14)}

\rowcolors{2}{white}{lightbg}
\begin{longtable}{@{}L{4.0cm}L{11.0cm}@{}}
\toprule
\rowcolor{primary!15}
\textbf{Quantity} & \textbf{Formula} \\
\midrule
\endhead
Information gain & $\text{Gain}(S, A) = H(S) - \sum_v \frac{|S_v|}{|S|}H(S_v)$ \\
Split information, gain ratio & $-\sum_v \frac{|S_v|}{|S|}\log_2\frac{|S_v|}{|S|}$; \quad Gain / SplitInfo \\
Gini index & $1 - \sum_i p_i^2$ \\
MAP hypothesis & $h_{MAP} = \arg\max_h P(D \mid h)P(h)$ \\
Naive Bayes & $v_{NB} = \arg\max_v P(v)\prod_j P(a_j \mid v)$ \\
Laplace smoothing & $(n_{v,c} + 1)/(n_c + k)$ \\
$k$-NN weighting & weight $= 1/d^2$ (or $1/d$) \\
Minkowski distance & $\big(\sum_j |x_j - z_j|^p\big)^{1/p}$ \\
SVM margin & $2/\|\mathbf{w}\|$ with $y_i(\mathbf{w}\cdot\mathbf{x}_i + b) \ge 1$ \\
Hinge loss & $\max(0,\ 1 - y f(\mathbf{x}))$ \\
Kernels & $(\mathbf{x}\cdot\mathbf{z} + r)^p$; \quad $\exp(-\gamma\|\mathbf{x} - \mathbf{z}\|^2)$ \\
Perceptron rule & $w_i \leftarrow w_i + \eta(t - o)x_i$ \\
Delta rule & $w_i \leftarrow w_i + \eta\sum_d(t_d - o_d)x_{id}$ \\
Backpropagation & $\delta_k = o_k(1 - o_k)(t_k - o_k)$; \quad $\delta_h = o_h(1 - o_h)\sum_k w_{kh}\delta_k$; \quad $\Delta w_{ji} = \eta\,\delta_j x_{ji}$ \\
Majority vote & $P(\text{correct}) = \sum_{k > n/2}\binom{n}{k}p^k(1 - p)^{n - k}$ \\
AdaBoost & $\alpha_t = \tfrac{1}{2}\ln\frac{1 - \varepsilon_t}{\varepsilon_t}$; mistakes $\times e^{\alpha_t}$, correct $\times e^{-\alpha_t}$, renormalise \\
\bottomrule
\end{longtable}

\section{Unsupervised, Sequences and Decisions (Chapters 15--21)}

\rowcolors{2}{white}{lightbg}
\begin{longtable}{@{}L{4.0cm}L{11.0cm}@{}}
\toprule
\rowcolor{primary!15}
\textbf{Quantity} & \textbf{Formula} \\
\midrule
\endhead
WCSS & $\sum_j\sum_{\mathbf{x} \in C_j}\|\mathbf{x} - \boldsymbol{\mu}_j\|^2$ \\
Silhouette & $s = (b - a)/\max(a, b)$ \\
PCA & eigenvectors of the covariance matrix; explained variance $\lambda_j/\sum_i\lambda_i$ \\
SOM update & $\mathbf{w}_u \leftarrow \mathbf{w}_u + \eta\,h(u)(\mathbf{x} - \mathbf{w}_u)$, \quad $h(u) = \exp(-d_{\text{grid}}^2/2\sigma^2)$ \\
EM responsibility & $r_{ij} = \pi_j\mathcal{N}(\mathbf{x}_i \mid \boldsymbol{\mu}_j, \Sigma_j)\big/\sum_l\pi_l\mathcal{N}(\mathbf{x}_i \mid \boldsymbol{\mu}_l, \Sigma_l)$ \\
EM M-step (means) & $\boldsymbol{\mu}_j = \sum_i r_{ij}\mathbf{x}_i/\sum_i r_{ij}$ \\
HMM forward & $\alpha_1(i) = \pi_i B_i(o_1)$; \quad $\alpha_{t+1}(j) = [\sum_i\alpha_t(i)A_{ij}]B_j(o_{t+1})$ \\
Viterbi & $\delta_{t+1}(j) = \max_i[\delta_t(i)A_{ij}]B_j(o_{t+1})$, with backpointers \\
Return & $G_t = \sum_k \gamma^k r_{t+k+1}$ \\
Bellman optimality & $V^{*}(s) = \max_a\big[R(s, a) + \gamma\sum_{s'}P(s' \mid s, a)V^{*}(s')\big]$ \\
Incremental mean & $Q \leftarrow Q + \tfrac{1}{n}(r - Q)$ \\
UCB & $Q(a) + c\sqrt{\ln t / N(a)}$ \\
TD(0) & $V(s) \leftarrow V(s) + \alpha[r + \gamma V(s') - V(s)]$ \\
Q-learning & $Q(s, a) \leftarrow Q(s, a) + \alpha[r + \gamma\max_{a'}Q(s', a') - Q(s, a)]$ \\
SARSA & $Q(s, a) \leftarrow Q(s, a) + \alpha[r + \gamma Q(s', a') - Q(s, a)]$ \\
Selection-rate ratio & $P(\hat{y} = 1 \mid B)/P(\hat{y} = 1 \mid A)$; four-fifths rule: flag if below 0.8 \\
\bottomrule
\end{longtable}
