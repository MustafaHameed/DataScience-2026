\chapter{Glossary}\label{app:glossary}
%=====================================================================
\normalfigures

\begin{multicols}{2}
\small
\raggedright

\textbf{adjacency list.} A graph representation storing each vertex's
neighbours. $\bigTh{|V|+|E|}$ space. (\chref{graphs})

\textbf{adjacency matrix.} A $|V| \times |V|$ array of edge indicators.
$\bigTh{|V|^{2}}$ space, $\bigTh{1}$ edge test. (\chref{graphs})

\textbf{amortised cost.} The total cost of a worst-case sequence of $n$
operations, divided by $n$. A worst-case quantity involving no probability ---
not an average case. (\chref{amortised})

\textbf{approximation algorithm.} One returning a solution provably within a
stated factor of optimal. (\chref{complexity})

\textbf{asymptotic notation.} $O$, $\Omega$, $\Theta$, $o$, $\omega$:
statements about how a function grows, ignoring constants and small inputs.
(\chref{asymptotics})

\textbf{average case.} Expected cost over a \emph{stated distribution of
inputs}. Meaningless until the distribution is named. (\chref{cases})

\textbf{best case.} The minimum cost over inputs of a given size. Promises
nothing. (\chref{cases})

\textbf{Bellman--Ford.} Single-source shortest paths by relaxing every edge
$|V|-1$ times. $\bigO{|V||E|}$; handles negative edges and detects negative
cycles. (\chref{shortestpaths})

\textbf{big-O.} $f = \bigO{g}$ if $\exists c, n_{0}$ with
$f(n) \le c\,g(n)$ for all $n \ge n_{0}$. An upper bound, not necessarily
tight. (\chref{asymptotics})

\textbf{big-Omega.} $f = \bigOm{g}$: a lower bound. Desirable about a problem,
undesirable about your algorithm. (\chref{asymptotics})

\textbf{big-Theta.} $f = \bigTh{g}$: both bounds. Equivalently
$\bigO{g} \cap \bigOm{g}$. (\chref{asymptotics})

\textbf{branch misprediction.} A processor's penalty for guessing a
conditional's outcome wrongly. Invisible to the RAM model; measured at a
factor of 2.0 in \chref{mergesort}.

\textbf{brute force.} Generating and testing every candidate. The baseline, the
oracle, and sometimes the answer. (\chref{bruteforce})

\textbf{certificate.} A short proof that an instance is a yes-instance,
checkable in polynomial time. What defines NP. (\chref{complexity})

\textbf{comparison sort.} A sort learning about its input only by comparing
pairs. Subject to the $\bigOm{n\lg n}$ bound. (\chref{lowerbound})

\textbf{counting sort.} A linear-time sort for integer keys in a small known
range. Outside the comparison bound because it never compares.
(\chref{lowerbound})

\textbf{cut property.} The lightest edge crossing any cut uncrossed by the
current partial tree is safe to add. The basis of both MST algorithms.
(\chref{mst})

\textbf{decision problem.} A problem with a yes/no answer. The form complexity
theory is built on. (\chref{complexity})

\textbf{decision tree.} A model of a comparison sort's execution; its leaves
are permutations and its height is the worst-case comparison count.
(\chref{lowerbound})

\textbf{dense.} $|E| = \bigTh{|V|^{2}}$. (\chref{graphs})

\textbf{Dijkstra's algorithm.} Single-source shortest paths, greedy, requiring
non-negative weights. (\chref{shortestpaths})

\textbf{divide and conquer.} Divide into smaller instances of the same problem,
conquer recursively, combine. (\chref{mergesort})

\textbf{dynamic programming.} Solving each distinct subproblem once and storing
the answer. Needs optimal substructure \emph{and} overlapping subproblems.
(\chref{dpone})

\textbf{edit distance.} The minimum number of insertions, deletions and
substitutions turning one string into another. (\chref{dptwo})

\textbf{exchange argument.} A proof that an optimal solution can be modified to
contain a greedy choice without getting worse. (\chref{greedy})

\textbf{expected running time.} The average over an algorithm's \emph{own
random choices}, for a fixed input. Assumes nothing about the data.
(\chref{cases}, \chref{quicksort})

\textbf{Floyd--Warshall.} All-pairs shortest paths in $\bigTh{|V|^{3}}$, with
the subproblem index outermost. (\chref{shortestpaths})

\textbf{greedy algorithm.} Commits to the locally best choice and never
reconsiders. Optimal when the greedy-choice property and optimal substructure
both hold. (\chref{greedy})

\textbf{greedy-choice property.} Some optimal solution contains the greedy
first choice. (\chref{greedy})

\textbf{hash flooding.} An attack supplying keys chosen to collide, turning a
hash table into a linked list. Measured at 324$\times$ in \chref{hashing}.

\textbf{heap.} A complete binary tree with the heap property, stored in an
array. Height exactly $\lfloor\lg n\rfloor$, guaranteed by the layout.
(\chref{structures})

\textbf{Huffman code.} An optimal prefix code built by repeatedly merging the
two least frequent symbols. (\chref{greedy})

\textbf{input size.} How much input there is: elements for an array, two
parameters for a graph, \emph{bits} for a number. (\chref{cases})

\textbf{inverse Ackermann, $\alpha(n)$.} A function below 5 for every writable
$n$; the amortised cost of a union--find operation. (\chref{mst})

\textbf{KMP.} Knuth--Morris--Pratt string matching, $\bigTh{n+m}$ always,
because the text pointer never moves backwards. (\chref{strings})

\textbf{load factor, $\alpha$.} $n/m$ in a hash table: the average chain
length. (\chref{hashing})

\textbf{loop invariant.} An assertion true before every iteration. Proved by
initialisation, maintenance and termination. (\chref{invariants})

\textbf{loop variant.} An integer-valued measure that strictly decreases,
proving termination. (\chref{invariants})

\textbf{master theorem.} Solves $T(n) = aT(n/b)+f(n)$ by comparing $f(n)$ with
$n^{\log_{b}a}$. (\chref{master})

\textbf{median of medians.} A deterministic worst-case linear selection
algorithm, with a constant poor enough that it is used as a fallback.
(\chref{selection})

\textbf{memoisation.} Caching a recursive function's results. Top-down dynamic
programming. (\chref{dpone})

\textbf{minimum spanning tree.} A spanning tree of least total weight.
(\chref{mst})

\textbf{NP.} Decision problems whose yes-instances have polynomial-time
verifiable certificates. \emph{Nondeterministic} polynomial, not ``not
polynomial''. (\chref{complexity})

\textbf{NP-complete.} In NP and NP-hard. A polynomial algorithm for any one of
them would give one for all of NP. (\chref{complexity})

\textbf{NP-hard.} At least as hard as every problem in NP. Need not be in NP.
(\chref{complexity})

\textbf{optimal substructure.} An optimal solution contains optimal solutions
to subproblems. Needed by both greedy and dynamic programming.
(\chref{greedy}, \chref{dpone})

\textbf{order statistic.} The $i$-th smallest element. (\chref{selection})

\textbf{overlapping subproblems.} A recursion revisits the same subproblems
repeatedly. What distinguishes dynamic programming from divide-and-conquer.
(\chref{dpone})

\textbf{P.} Decision problems solvable in polynomial time.
(\chref{complexity})

\textbf{partition.} Rearranging an array about a pivot. Quicksort's divide
step. (\chref{quicksort})

\textbf{path compression.} Flattening the path to the root during a
union--find \texttt{find}. (\chref{mst})

\textbf{potential method.} Amortised analysis via a function $\Phi$ of the
state, with $\hat c = c + \Delta\Phi$. (\chref{amortised})

\textbf{prefix code.} A code in which no codeword is a prefix of another.
(\chref{greedy})

\textbf{prefix function.} For KMP: the longest proper prefix of
$P[1..i]$ that is also a suffix. (\chref{strings})

\textbf{Prim's algorithm.} MST by growing one tree, taking the lightest edge
leaving it. (\chref{mst})

\textbf{pseudo-polynomial.} Polynomial in the numeric \emph{value} of an input
number, exponential in its bit \emph{length}. Knapsack's $\bigTh{nW}$.
(\chref{dptwo}, \chref{complexity})

\textbf{quickselect.} Quicksort with one recursive call deleted. $\bigTh{n}$
expected. (\chref{selection})

\textbf{Rabin--Karp.} String matching by rolling hash. $\bigO{n+m}$ expected;
inherits every caveat of hashing. (\chref{strings})

\textbf{radix sort.} Counting sort applied digit by digit. Requires a stable
inner sort. (\chref{lowerbound})

\textbf{RAM model.} Unit-cost random-access machine: one operation, one time
unit, uniform memory. All three assumptions are false and all three are
constant factors. (\chref{role})

\textbf{randomised algorithm.} One making its own random choices, so its
guarantee holds for every input. (\chref{quicksort})

\textbf{recurrence.} An equation defining a function in terms of itself at
smaller arguments. (\chref{recurrences})

\textbf{recursion tree.} A device for \emph{guessing} a recurrence's solution
by summing level by level. Not a proof. (\chref{recurrences})

\textbf{reduction.} $A \le_{P} B$: a polynomial-time transformation preserving
yes-instances. Means $B$ is at least as hard as $A$. (\chref{complexity})

\textbf{regularity condition.} $af(n/b) \le cf(n)$ for some $c<1$; required by
case 3 of the master theorem. (\chref{master})

\textbf{simple uniform hashing.} The assumption that keys scatter evenly ---
about the hash \emph{and} the keys together. (\chref{hashing})

\textbf{sparse.} $|E| = \bigO{|V|}$. (\chref{graphs})

\textbf{stability.} A sort is stable if equal keys keep their input order.
Required by radix sort's inner pass. (\chref{mergesort},
\chref{lowerbound})

\textbf{strengthening the hypothesis.} Proving a stronger statement so the
induction has more to work with --- subtracting a lower-order term.
(\chref{recurrences})

\textbf{substitution method.} Guess the solution to a recurrence, then prove it
by induction. (\chref{recurrences})

\textbf{union by rank.} Attaching the shorter union--find tree under the
taller. (\chref{mst})

\textbf{watershed function.} $n^{\log_{b}a}$: the leaf count of the recursion
tree, against which $f(n)$ is compared. (\chref{master})

\textbf{worst case.} The maximum cost over inputs of a given size. A
guarantee, and the default unless stated otherwise. (\chref{cases})

\end{multicols}
