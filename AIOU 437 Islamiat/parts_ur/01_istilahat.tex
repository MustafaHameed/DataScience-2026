\section*{حصہ اول --- مختصر اصطلاحات}
\markboth{مختصر اصطلاحات}{}

\noindent
بہار \textenglish{2013} اور خزاں \textenglish{2021} دونوں میں \textbf{پہلا
سوال یہی تھا اور لازمی تھا}: دس اصطلاحات، فی اصطلاح دو نمبر، کل بیس نمبر ---
اور صرف بیس منٹ کا سوال۔ لمبا جواب نہ لکھیے؛ ایک سطر تعریف اور ایک حوالہ کافی
ہے۔ حوالہ (آیت، حدیث یا سنہ) ہی دوسرا نمبر دلاتا ہے۔

\medskip
\begin{center}
\begin{tabular}{@{}p{2.9cm} p{11.5cm}@{}}
\toprule
\textbf{محکمات و متشابہات}
  & واضح المعنی آیات (اصل کتاب) اور جن کا حقیقی مفہوم اللہ کے سوا کوئی نہیں
    جانتا۔ \ar{هُنَّ أُمُّ الْكِتَابِ وَأُخَرُ مُتَشَابِهَاتٌ} (آل عمران،
    \textenglish{7})۔ {\small\color{bmgrey}[خزاں \textenglish{2021}]} \\
\textbf{حدیث قدسی}
  & مضمون اللہ کی طرف سے، الفاظ نبی کریم \saw کے۔ قرآن سے فرق: قرآن لفظاً و
    معناً منزل، متواتر اور نماز میں پڑھا جاتا ہے؛ حدیث قدسی نہیں۔
    {\small\color{bmgrey}[بہار \textenglish{2013}]} \\
\textbf{صحاح ستہ}
  & بخاری، مسلم، ابن ماجہ، ابو داؤد، ترمذی، نسائی --- حدیث کی چھ مستند کتابیں؛
    پہلی دو ''صحیحین''۔ {\small\color{bmgrey}[بہار \textenglish{2013}]} \\
\textbf{تقدیر مبرم و معلق}
  & \textbf{مبرم} اٹل فیصلہ؛ \textbf{معلق} دعا، صدقہ یا صلہ رحمی سے بدل سکتا
    ہے۔ ''\ar{لَا يَرُدُّ الْقَضَاءَ إِلَّا الدُّعَاءُ}'' (ترمذی)۔
    {\small\color{bmgrey}[دونوں پرچوں میں]} \\
\textbf{کبر}
  & ''\ar{الْكِبْرُ بَطَرُ الْحَقِّ وَغَمْطُ النَّاسِ}'' --- حق کو ٹھکرانا،
    لوگوں کو حقیر جاننا (مسلم)۔ {\small\color{bmgrey}[بہار \textenglish{2013}]} \\
\textbf{حسن خلق}
  & ''\ar{الْبِرُّ حُسْنُ الْخُلُقِ}'' --- نیکی خوش خلقی ہی کا نام ہے (مسلم)۔
    {\small\color{bmgrey}[بہار \textenglish{2013}]} \\
\textbf{الحب فی اللہ}
  & محض اللہ کی رضا کے لیے محبت --- نسب، مال یا مفاد کی بنا پر نہیں۔ تفصیل
    سوال \textenglish{9} میں۔ {\small\color{bmgrey}[دونوں پرچوں میں]} \\
\textbf{رحمتِ الٰہی کی حد}
  & رحمت وہاں تک ہے جہاں اللہ کی حد شروع ہوتی ہے: ''\ar{أَتَشْفَعُ فِي حَدٍّ
    مِنْ حُدُودِ اللهِ؟}'' (بخاری و مسلم)۔ {\small\color{bmgrey}[خزاں
    \textenglish{2021}]} \\
\textbf{اتباع و اطاعت کا فرق}
  & اطاعت حکم کی تعمیل؛ اتباع قدم بہ قدم پیروی۔ \ar{قُلْ إِن كُنتُمْ
    تُحِبُّونَ اللهَ فَاتَّبِعُونِي} (آل عمران، \textenglish{31})۔
    {\small\color{bmgrey}[خزاں \textenglish{2021}]} \\
\bottomrule
\end{tabular}
\end{center}
